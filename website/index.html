A Tessel library

Web servers in a few lines of Tessel.

Routes with values in the path, middleware, forms, uploads, cookies, signed sessions, folders of static files and HTML templates. It's built on Tessel's own HTTP server, with nothing else to install.

tessel add web
See examples Code on GitHub

MIT licence · Tessel 0.23.1 or later · no dependencies

main.tsl
import web

fn main() {
    var router = web.Router()
    router.get("/hello/:name") { r in
        web.json(["greeting": "Hello, {r.param("name")}!"])
    }
    web.serve(router, port: 8080)
}
↓ curl -i 127.0.0.1:8080/hello/Ann%20Lee
Response

Content-Length: 30

Examples

Each program below was run with the library, and the output below it is what it printed or what curl got back. app is a word Tessel keeps for apps, so the router is called router.

A JSON API

:id takes one part of the path. web.json sends any value as JSON, and web.readJson reads a body into your type, or gives nil if it isn't JSON of that shape. A path that only other methods answer gets a 405 with an Allow header.

import web

struct Item {
    id: Int
    name: String
}

fn main() {
    var items = [Item(id: 1, name: "Tea")]

    var router = web.Router()
    router.use(web.logger())
    router.get("/items") { r in web.json(items) }
    router.get("/items/:id") { r in
        for item in items {
            if "{item.id}" == r.param("id") {
                return web.json(item)
            }
        }
        web.status(404)
    }
    router.post("/items") { r in
        let sent: Item? = web.readJson(r)
        if let item = sent {
            items.append(item)
            return web.json(item, status: 201)
        }
        web.status(400, "send an item as JSON")
    }
    web.serve(router, port: 8080)
}

Forms, redirects and uploads

r.form() reads a form, urlencoded or multipart. After a form, answer with a 303 so a reload doesn't send it again. r.file(name) gives an uploaded file with its name, type and bytes.

import web

fn main() {
    var names: [String] = []

    var router = web.Router()
    router.get("/") { r in
        let error = r.queryValue("error") ?? ""
        web.text("Signed by: {names.joined(separator: ", ")}\n{error}")
    }
    router.post("/sign") { r in
        let name = (r.form()["name"] ?? "").trim()
        if name.isEmpty {
            return web.redirect("/?error=Write+a+name", status: 303)
        }
        names.append(name)
        web.redirect("/", status: 303)
    }
    router.post("/upload") { r in
        if let f = r.file("photo") {
            return web.text("{f.fileName ?? "?"}: {f.data.count} bytes of {f.contentType}")
        }
        web.status(400, "no photo")
    }
    web.serve(router, port: 8080)
}

Signing in with a session

The session's values are kept in a cookie signed with HMAC-SHA256, so they can't be changed without the secret. They can be read, so never put a password in them. The secret must be at least 16 characters; keep it out of the code.

import web

fn main() {
    var router = web.Router()
    router.use(web.sessions(secret: environment("SESSION_SECRET") ?? ""))
    router.post("/login") { r in
        let form = r.form()
        if form["password"] != "tessel" {
            return web.redirect("/?wrong", status: 303)
        }
        web.redirect("/", status: 303).withSession(["user": form["name"] ?? ""])
    }
    router.get("/") { r in web.text("Hello, {r.session["user"] ?? "stranger"}") }
    router.post("/logout") { r in web.redirect("/", status: 303).withSession([:]) }
    web.serve(router, port: 8080)
}

Static files

Each file goes with its content type, ETag and Last-Modified. A browser that has the file already gets a 304 and no bytes, and a Range request gets only those bytes. Hidden files and names with .. aren't sent.

import web

fn main() {
    var router = web.Router()
    router.files("/static", folder: "public")
    web.serve(router, port: 8080)
}

Templates with a layout

Templates are written like Handlebars templates and give what Handlebars gives. Values are escaped for HTML; {{{body}}} puts the page into the layout as it is. In a handler, web.page(pages, "shop", shop, layout: "layout") is the same page as an answer.

views/shop.html

<h1>{{title}}</h1>
<ul>
  {{#each items}}
  <li>{{name}}: ${{price}}{{#if @first}} (new){{/if}}</li>
  {{else}}
  <li>Nothing for sale</li>
  {{/each}}
</ul>

views/layout.html

<!doctype html>
<title>{{title}}</title>
{{{body}}}

main.tsl

import web

struct Item {
    name: String
    price: Float
}

struct Shop {
    title: String
    items: [Item]
}

fn main() {
    let pages = try web.loadTemplates("views")
    let shop = Shop(title: "Tea & <coffee>", items: [Item(name: "Green tea", price: 3.5), Item(name: "Espresso", price: 2.25)])
    print(web.render(pages, "shop", shop, layout: "layout"))
    print(web.render(pages, "shop", Shop(title: "Closed", items: [])))
}

Middleware of your own

A middleware's before passes a request on with web.next or answers it with web.stop; its after can change each answer. The befores run in the order added, the afters in the opposite order.

import web

fn main() {
    var router = web.Router()
    router.use(web.onRequest { r in
        if r.header("x-api-key") != "secret" {
            return web.stop(web.status(401))
        }
        web.next(r)
    })
    router.use(web.onResponse { r, res in res.withHeader("X-Served-By", "web") })
    router.get("/report") { r in web.text("all good") }
    web.serve(router, port: 8080)
}

Testing without a network

web.call(router, method, path, body:, headers:) runs a request through a router and gives the answer, so a test needs no server and no port.

import web

fn routes() -> web.Router {
    var router = web.Router()
    router.get("/hello/:name") { r in web.text("Hello, {r.param("name")}!") }
    router
}

test "greets by name" {
    let answer = web.call(routes(), "GET", "/hello/Ann%20Lee")
    expect(answer?.status == 200)
    expect(answer?.text() == "Hello, Ann Lee!")
}

test "other paths are not found" {
    expect(web.call(routes(), "GET", "/bye")?.status == 404)
}

Reference

The README has all of it, with more detail.

Router

Method
get, post, put, patch, delete(pattern) { r in … }Answers requests with that method whose path fits the pattern. A get route answers HEAD too.
any(pattern) { … }Any method.
add(method, pattern) { … }Another method ("PROPFIND").
later(method, pattern) { r, reply in … }A handler that calls reply when the answer is ready: after a request of its own, a timer or background work.
use(middleware)Middleware for every request, in the order added.
mount(prefix, router)Another router's routes under a prefix. Its middleware runs only for its routes.
files(prefix, folder:, options:)A folder of files. FileOptions has index, maxAge and dotFiles.
notFound { r in … }What answers when no route fits.

A pattern's parts can be :name (one part of the path) or, at the end, *name (the rest of the path). Fixed text wins over :name, and :name over *name, whatever order the routes were added in. web.serve(router, port:, host:) starts the server and returns its id, or -1 if it can't listen. It listens on this computer only unless host: "0.0.0.0", and speaks HTTP/1.1, not HTTPS: on the internet, put it behind a server that does HTTPS.

Requests

Field or method
method, path, query"GET", "/users/3", "page=2". The path is tidied as the router sees it.
headers, header(name)Names in lowercase; header takes any case.
body, text()The body as Data, and as text.
param(name), paramsValues from the pattern, decoded.
queryValue(name), queryValues(name), queryDict()The query, decoded.
form()The fields of a form: urlencoded or multipart.
parts(), file(name)The parts of a multipart body, and an uploaded file: name, fileName, contentType, data.
cookies(), cookie(name)The cookies the browser sent.
accepts(type), contentType()Whether the Accept header takes a type; the body's type without parameters.
values, sessionNotes left by middleware, and the session's values.
remote, receivedThe client's address, and when the request came.
web.readJson(r)The body as JSON in your type, or nil.

Responses

Function
web.text(text, status:)Plain text.
web.html(html, status:)A page.
web.json(value, status:)Any value as JSON: a struct, a list, a dictionary.
web.bytes(data, type:, status:)Bytes with a content type.
web.redirect(to, status:)302 by default; 303 after a form, 301 or 308 for good.
web.status(code, message)A status with its reason as the body, or a message.
web.file(path, request, maxAge:, download:)A file, with caching and ranges.
web.page(templates, name, data, layout:)A page from templates.

A web.Response (status, headers, body) changes with withHeader, withStatus, withCookie(name, value, maxAge:, path:, domain:, httpOnly:, secure:, sameSite:), withoutCookie and withSession. Cookies are HttpOnly and SameSite=Lax unless you say otherwise.

Middleware

Middleware
logger()Prints a line for each request: address, method, path, status, size and time.
cors(origins:, methods:, headers:, credentials:, maxAge:)Lets pages of other sites call the server, and answers preflight requests.
basicAuth(realm:) { user, password in … }Asks for a name and password; the name goes in values["user"]. Only over HTTPS.
sessions(secret:, cookie:, maxAge:, secure:, sameSite:)Sessions in a signed cookie, two weeks from when last saved.
compress(minimum:)gzip for text, HTML, CSS, JavaScript, JSON, XML and SVG, for clients that take it.
securityHeaders(contentSecurityPolicy:)nosniff, no framing, a strict referrer policy, and a CSP if given.
bodyLimit(bytes)A 413 for bigger bodies.
onRequest, onResponse, Middleware(before:, after:)Middleware of your own.

web.sign, web.unsign and web.hmac sign values of your own.

Template tags

Tag
{{name}}A value, with HTML's special characters escaped.
{{{html}}}, {{& html}}A value as it is.
{{user.name}}, {{items.0}}, {{items.count}}A field, an item, a count.
{{#if x}} … {{else if y}} … {{else}} … {{/if}}False, nil, 0, "" and an empty list are false.
{{#unless x}} … {{/unless}}The opposite.
{{#each items}} … {{else}} … {{/each}}Each item, with {{this}}, {{@index}}, {{@first}}, {{@last}}, {{@key}}.
{{#each items as |item i|}}With names for the item and its index.
{{#with user}} … {{/with}}Names looked up in a value.
{{> row}}, {{> row item}}Another template, with the same value or another.
{{! note }}, {{!-- note --}}A comment.
{{~name~}}Takes away the spaces and line breaks on that side.
{{@root.title}}, ../nameA name from the top, or one level out.
\{{name}}The text {{name}}.

There are no helpers: work such values out in Tessel and pass them in. web.loadTemplates(folder, extension:) reads a folder and its subfolders; web.parseTemplates(["name": text]) makes templates from texts. In a Tessel string, { starts a value, so a template written in the code is "\{\{name}}".
